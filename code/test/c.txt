package main

import (
    "fmt"
    "sync"
    "time"
)

func square(num int, wg *sync.WaitGroup, ch chan int) {
    defer wg.Done()
    ch <- num * num
}

func main() {
    numbers := []int{1, 2, 3, 4, 5}
    ch := make(chan int, len(numbers))
    var wg sync.WaitGroup

    for _, n := range numbers {
        wg.Add(1)
        go square(n, &wg, ch)
    }

    wg.Wait()
    close(ch)

    sum := 0
    for sq := range ch {
        sum += sq
    }

    fmt.Printf("平方和为: %d\n", sum)
}
