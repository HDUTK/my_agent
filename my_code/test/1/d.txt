using System;

class Fibonacci
{
    static void Main()
    {
        Console.Write("请输入要生成的斐波那契数列项数：");
        int n = int.Parse(Console.ReadLine());

        int[] fib = new int[n];
        if (n >= 1) fib[0] = 0;
        if (n >= 2) fib[1] = 1;

        for (int i = 2; i < n; i++)
        {
            fib[i] = fib[i - 1] + fib[i - 2];
        }

        Console.WriteLine($"斐波那契数列前 {n} 项：");
        for (int i = 0; i < n; i++)
        {
            Console.Write(fib[i] + " ");
        }
    }
}
